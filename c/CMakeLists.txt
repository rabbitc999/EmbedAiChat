cmake_minimum_required(VERSION 3.20)
project(EmbedAiChatC VERSION 0.1.0 LANGUAGES C)
set(EMBED_AI_C_PROFILE "desktop" CACHE STRING "desktop or mcu (STM32F4/ESP32-C3)")
set_property(CACHE EMBED_AI_C_PROFILE PROPERTY STRINGS desktop mcu)
if(EMBED_AI_C_PROFILE STREQUAL "mcu")
 set(profile_mcu 1)
 set(desktop_default OFF)
else()
 if(NOT EMBED_AI_C_PROFILE STREQUAL "desktop")
  message(FATAL_ERROR "EMBED_AI_C_PROFILE must be desktop or mcu")
 endif()
 set(profile_mcu 0)
 set(desktop_default ON)
endif()
option(EMBED_AI_C_CURL "Build desktop curl adapter and console" ${desktop_default})
option(EMBED_AI_C_MCP "Build desktop stdio MCP adapter (not for MCU)" ${EMBED_AI_C_CURL})
option(EMBED_AI_C_TOOLS "Host tool callbacks and schema validation" ON)
option(EMBED_AI_C_SKILLS "Skill loading and activation" ${desktop_default})
option(EMBED_AI_C_ARCHIVE "JSON history import/export" ${desktop_default})
option(EMBED_AI_C_HISTORY "Retain earlier conversation messages" ON)
option(EMBED_AI_C_COMPACTION "Rolling summary" ${desktop_default})
option(EMBED_AI_C_ANTHROPIC "Anthropic protocol" ${desktop_default})
option(EMBED_AI_C_AUDIT "Retain executed tool audit" ON)
if(EMBED_AI_C_PROFILE STREQUAL "mcu" AND (EMBED_AI_C_CURL OR EMBED_AI_C_MCP))
 message(FATAL_ERROR "MCU profile must disable CURL and desktop stdio MCP; use a fresh build directory")
endif()
if(EMBED_AI_C_MCP AND NOT EMBED_AI_C_TOOLS)
 message(FATAL_ERROR "Desktop MCP requires tools")
endif()
add_library(embed_ai_chat_c src/ai_chat.c src/hal_net.c vendor/cjson/cJSON.c)
target_compile_definitions(embed_ai_chat_c PUBLIC AI_CHAT_MCU=${profile_mcu})
foreach(feature TOOLS SKILLS ARCHIVE HISTORY COMPACTION ANTHROPIC AUDIT)
 target_compile_definitions(embed_ai_chat_c PUBLIC AI_CHAT_ENABLE_${feature}=$<BOOL:${EMBED_AI_C_${feature}}>)
endforeach()
foreach(limit MAX_JSON_BYTES DEFAULT_RESPONSE_BYTES DEFAULT_CONTEXT_BUDGET MIN_CONTEXT_BUDGET DEFAULT_MAX_TOKENS DEFAULT_SUMMARY_TOKENS MAX_TOOL_RESULT_BYTES MAX_SKILL_BYTES MAX_AUDIT_ENTRIES MAX_HISTORY_MESSAGES)
 set(EMBED_AI_C_${limit} "" CACHE STRING "Override AI_CHAT_${limit} (empty uses profile default)")
 if(NOT "${EMBED_AI_C_${limit}}" STREQUAL "")
  if(NOT "${EMBED_AI_C_${limit}}" MATCHES "^[0-9]+$")
   message(FATAL_ERROR "EMBED_AI_C_${limit} must be an unsigned integer")
  endif()
  target_compile_definitions(embed_ai_chat_c PUBLIC AI_CHAT_${limit}=${EMBED_AI_C_${limit}})
 endif()
endforeach()
if(profile_mcu)
 target_compile_definitions(embed_ai_chat_c PRIVATE CJSON_NESTING_LIMIT=32)
 if(NOT MSVC)
  target_compile_options(embed_ai_chat_c PRIVATE -Os -ffunction-sections -fdata-sections)
 endif()
endif()
target_include_directories(embed_ai_chat_c PUBLIC include vendor/cjson)
set_target_properties(embed_ai_chat_c PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
if(MSVC)
 target_compile_options(embed_ai_chat_c PUBLIC /utf-8)
else()
 target_compile_options(embed_ai_chat_c PRIVATE -Wall -Wextra -Wpedantic)
 target_link_libraries(embed_ai_chat_c PUBLIC m)
endif()
if(EMBED_AI_C_MCP)
 find_package(Threads REQUIRED)
 add_library(embed_ai_chat_c_mcp src/mcp_stdio.c)
 target_link_libraries(embed_ai_chat_c_mcp PUBLIC embed_ai_chat_c Threads::Threads)
 set_target_properties(embed_ai_chat_c_mcp PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
 install(TARGETS embed_ai_chat_c_mcp)
endif()
if(EMBED_AI_C_CURL)
 find_package(CURL REQUIRED)
 add_library(embed_ai_chat_c_curl hal/hal_net_curl.c)
 target_link_libraries(embed_ai_chat_c_curl PUBLIC embed_ai_chat_c CURL::libcurl)
 add_executable(EmbedAiChatC src/main.c)
 target_link_libraries(EmbedAiChatC PRIVATE embed_ai_chat_c_curl)
 target_compile_definitions(EmbedAiChatC PRIVATE AI_CHAT_DESKTOP_MCP=$<BOOL:${EMBED_AI_C_MCP}>)
 if(EMBED_AI_C_MCP)
  target_link_libraries(EmbedAiChatC PRIVATE embed_ai_chat_c_mcp)
 endif()
endif()
include(CTest)
if(BUILD_TESTING)
 if(EMBED_AI_C_SKILLS AND EMBED_AI_C_TOOLS AND EMBED_AI_C_ARCHIVE AND EMBED_AI_C_HISTORY AND EMBED_AI_C_ANTHROPIC AND EMBED_AI_C_AUDIT AND NOT profile_mcu)
  add_executable(EmbedAiChatCTests tests/test_core.c)
 else()
  add_executable(EmbedAiChatCTests tests/test_mcu.c)
 endif()
 target_link_libraries(EmbedAiChatCTests PRIVATE embed_ai_chat_c)
 add_test(NAME c_core_hal COMMAND EmbedAiChatCTests)
endif()
install(TARGETS embed_ai_chat_c)
install(DIRECTORY include/ DESTINATION include)
install(FILES vendor/cjson/cJSON.h DESTINATION include)
