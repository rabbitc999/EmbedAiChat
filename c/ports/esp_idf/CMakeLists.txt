# 把本目录加入 ESP-IDF EXTRA_COMPONENT_DIRS；使用 SDK 自带 json，避免重复 cJSON。
idf_component_register(
 SRCS "../../src/ai_chat.c" "../../src/hal_net.c" "hal_net_esp_idf.c"
 INCLUDE_DIRS "../../include" "."
 REQUIRES esp_http_client mbedtls json)
target_compile_definitions(${COMPONENT_LIB} PUBLIC AI_CHAT_MCU=1)
foreach(feature TOOLS HISTORY AUDIT COMPACTION ARCHIVE SKILLS ANTHROPIC)
 target_compile_definitions(${COMPONENT_LIB} PUBLIC AI_CHAT_ENABLE_${feature}=$<BOOL:${CONFIG_EMBED_AI_${feature}}>)
endforeach()
target_compile_definitions(${COMPONENT_LIB} PUBLIC
 AI_CHAT_MAX_JSON_BYTES=${CONFIG_EMBED_AI_JSON_BYTES}
 AI_CHAT_DEFAULT_RESPONSE_BYTES=${CONFIG_EMBED_AI_RESPONSE_BYTES}
 AI_CHAT_DEFAULT_CONTEXT_BUDGET=${CONFIG_EMBED_AI_CONTEXT_TOKENS}
 AI_CHAT_DEFAULT_MAX_TOKENS=${CONFIG_EMBED_AI_OUTPUT_TOKENS}
 AI_CHAT_DEFAULT_SUMMARY_TOKENS=${CONFIG_EMBED_AI_SUMMARY_TOKENS}
 AI_CHAT_MAX_TOOL_RESULT_BYTES=${CONFIG_EMBED_AI_TOOL_RESULT_BYTES}
 AI_CHAT_MAX_AUDIT_ENTRIES=${CONFIG_EMBED_AI_AUDIT_ENTRIES}
 AI_CHAT_MAX_HISTORY_MESSAGES=${CONFIG_EMBED_AI_HISTORY_MESSAGES})
target_compile_options(${COMPONENT_LIB} PRIVATE -Os)
