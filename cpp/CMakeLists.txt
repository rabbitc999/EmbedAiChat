cmake_minimum_required(VERSION 3.20)
project(EmbedAiChat VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(CURL REQUIRED)
find_package(Threads REQUIRED)
add_library(embed_ai_chat src/framework.cpp src/process.cpp src/json_util.cpp vendor/chali_json/json.cpp)
target_include_directories(embed_ai_chat PUBLIC include vendor)
target_link_libraries(embed_ai_chat PUBLIC CURL::libcurl Threads::Threads)
if(MSVC)
 target_compile_options(embed_ai_chat PUBLIC /utf-8)
 target_compile_definitions(embed_ai_chat PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
add_executable(EmbedAiChatCpp src/main.cpp)
target_link_libraries(EmbedAiChatCpp PRIVATE embed_ai_chat)
install(TARGETS embed_ai_chat EmbedAiChatCpp)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY vendor/chali_json DESTINATION include FILES_MATCHING PATTERN "*.hpp")

include(CTest)
if(BUILD_TESTING)
 add_executable(EmbedAiChatJsonTests ../tests/json_tests.cpp)
 target_link_libraries(EmbedAiChatJsonTests PRIVATE embed_ai_chat)
 add_test(NAME native_json COMMAND EmbedAiChatJsonTests)
endif()
